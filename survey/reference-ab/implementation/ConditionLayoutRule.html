10/** Normalize a short if condition after block wrapping, irrespective of the engine's rule ordering. */
11class ConditionLayoutRule : Rule(
12    RuleId("nur:compact-condition"), About(), usesEditorConfigProperties = setOf(MAX_LINE_LENGTH_PROPERTY),
13    visitorModifiers = setOf(VisitorModifier.RunAsLateAsPossible,
14        VisitorModifier.RunAfterRule(RuleId("nur:nested-lambda-wrapping"), VisitorModifier.RunAfterRule.Mode.ONLY_WHEN_RUN_AFTER_RULE_IS_LOADED_AND_ENABLED),
15        VisitorModifier.RunAfterRule(RuleId("standard:condition-wrapping"), VisitorModifier.RunAfterRule.Mode.ONLY_WHEN_RUN_AFTER_RULE_IS_LOADED_AND_ENABLED)),
16), RuleAutocorrectApproveHandler {
17    private var width = 140
18    override fun beforeFirstNode(editorConfig: EditorConfig) {
19        width = editorConfig[MAX_LINE_LENGTH_PROPERTY].let { if (it <= 0) Int.MAX_VALUE else it }
20    }
21    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
22        val expression = node.psi as? KtIfExpression ?: return
23        val condition = expression.condition ?: return
24        val right = expression.rightParenthesis ?: return
25        val left = expression.leftParenthesis ?: return
26        if (PsiTreeUtil.findChildrenOfType(expression, PsiComment::class.java).any {
27                it.textOffset > left.textOffset && it.textRange.endOffset <= right.textOffset
28            } ||
29            PsiTreeUtil.findChildOfType(condition, KtLambdaExpression::class.java) != null ||
30            PsiTreeUtil.findChildrenOfType(condition, KtStringTemplateExpression::class.java).any { '\n' in it.text }) return
31        val text = expression.containingFile.text
32        val start = text.lastIndexOf('\n', left.textOffset - 1) + 1
33        val flat = condition.text.replace(Regex("\\s+"), " ")
34        val endOfLine = text.indexOf('\n', right.textRange.endOffset).let { if (it < 0) text.length else it }
35        val tailLength = endOfLine - right.textRange.endOffset
36        if (left.textRange.endOffset - start + flat.length + 1 + tailLength > width) return
37        val spaces = mutableListOf<ASTNode>()
38        fun collect(n: ASTNode) {
39            if (n.elementType == ElementType.WHITE_SPACE && '\n' in n.text) spaces.add(n)
40            var child = n.firstChildNode
41            while (child != null) { collect(child); child = child.treeNext }
42        }
43        collect(condition.node)
44        for (space in spaces) emit(space.startOffset, "Keep a short if condition on one line", true).ifAutocorrectAllowed {
45            space.replaceTextWith(" ")
46        }
47        val before = condition.node.firstChildLeafOrSelf20.prevLeaf
48        if (before?.elementType == ElementType.WHITE_SPACE && '\n' in before.text) {
49            emit(before.startOffset, "Keep a short condition after the opening parenthesis", true).ifAutocorrectAllowed { before.remove() }
50        }
51        val previous = right.node.prevLeaf
52        if (previous?.elementType == ElementType.WHITE_SPACE && '\n' in previous.text) {
53            emit(previous.startOffset, "Keep the closing parenthesis with a short condition", true).ifAutocorrectAllowed { previous.remove() }
54        }
55    }
56}

34/** Separate multiline branches without inserting gaps between adjacent simple branches. */
35private fun blankFlag(name: String, default: Boolean) = com.pinterest.ktlint.rule.engine.core.api.editorconfig.EditorConfigProperty(
36    type = org.ec4j.core.model.PropertyType.LowerCasingPropertyType(name, name,
37        org.ec4j.core.model.PropertyType.PropertyValueParser.BOOLEAN_VALUE_PARSER, setOf("true", "false")), defaultValue = default,
38)
39private val AFTER_MULTILINE = blankFlag("nur_blank_after_multiline_when", true)
40private val AFTER_SIMPLE = blankFlag("nur_blank_after_simple_when", false)
41class WhenBranchSpacingRule : Rule(
42    RuleId("nur:when-branch-spacing"), About(),
43    usesEditorConfigProperties = setOf(AFTER_MULTILINE, AFTER_SIMPLE),
44    visitorModifiers = setOf(VisitorModifier.RunAsLateAsPossible),
45), RuleAutocorrectApproveHandler {
46    private var afterMultiline = true
47    private var afterSimple = false
48    override fun beforeFirstNode(editorConfig: EditorConfig) {
49        afterMultiline = editorConfig[AFTER_MULTILINE]
50        afterSimple = editorConfig[AFTER_SIMPLE]
51    }
52    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
53        val expression = node.psi as? KtWhenExpression ?: return
54        if ('\n' !in expression.text) return
55        val entries = expression.entries
56        for (index in 1 until entries.size) {
57            val entry = entries[index]
58            // Do not move or reinterpret comments between branches or semicolon-separated code.
59            val whitespace = entry.prevSibling as? PsiWhiteSpace ?: continue
60            if (whitespace.prevSibling != entries[index - 1] || '\n' !in whitespace.text) continue
61            val separated = if ('\n' in entries[index - 1].text) afterMultiline else afterSimple
62            val gap = if (separated) "\n\n" else "\n"
63            val desired = gap + whitespace.text.substringAfterLast('\n')
64            if (whitespace.text == desired) continue
65            emit(whitespace.textOffset, "Use one blank line after a multiline when branch and none after a simple branch", true)
66                .ifAutocorrectAllowed { whitespace.node.replaceTextWith(desired) }
67        }
68    }

3import com.pinterest.ktlint.rule.engine.core.api.*
4import com.pinterest.ktlint.rule.engine.core.api.editorconfig.EditorConfig
5import org.jetbrains.kotlin.com.intellij.lang.ASTNode
6import org.jetbrains.kotlin.com.intellij.psi.PsiComment
7import org.jetbrains.kotlin.com.intellij.psi.util.PsiTreeUtil
8import org.jetbrains.kotlin.psi.*

10/** Optional survey prototypes. Disabled by default; not installed in application projects. */
11class BallotLayoutOverridesRule : Rule(
12    RuleId("nur:ballot-layout-overrides"), About(),
13    usesEditorConfigProperties = setOf(BODY_FIRST_BLANK_LINE, PARAMETER_ANNOTATION_NEWLINE, ENUM_ANNOTATION_INLINE, EMPTY_CALL_PARENTHESES, EMPTY_LAMBDA_MULTILINE),
14    visitorModifiers = setOf(VisitorModifier.RunAsLateAsPossible,
15        VisitorModifier.RunAfterRule(RuleId("nur:class-layout"), VisitorModifier.RunAfterRule.Mode.REGARDLESS_WHETHER_RUN_AFTER_RULE_IS_LOADED_OR_DISABLED),
16        VisitorModifier.RunAfterRule(RuleId("standard:function-signature"), VisitorModifier.RunAfterRule.Mode.REGARDLESS_WHETHER_RUN_AFTER_RULE_IS_LOADED_OR_DISABLED),
17        VisitorModifier.RunAfterRule(RuleId("nur:nested-lambda-wrapping"), VisitorModifier.RunAfterRule.Mode.REGARDLESS_WHETHER_RUN_AFTER_RULE_IS_LOADED_OR_DISABLED)),
18), RuleAutocorrectApproveHandler {
19    private var emptyLambda = false
20    private var emptyParentheses = false
21    private var blankFirst = false
22    private var parameterAnnotation = false
23    private var enumAnnotation = false
24    override fun beforeFirstNode(editorConfig: EditorConfig) {
25        blankFirst = editorConfig[BODY_FIRST_BLANK_LINE]
26        emptyParentheses = editorConfig[EMPTY_CALL_PARENTHESES]
27        emptyLambda = editorConfig[EMPTY_LAMBDA_MULTILINE]
28        parameterAnnotation = editorConfig[PARAMETER_ANNOTATION_NEWLINE]
29        enumAnnotation = editorConfig[ENUM_ANNOTATION_INLINE]
30    }
31    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
32        val element = node.psi
33        if (element is KtLambdaExpression && emptyLambda && element.bodyExpression?.statements?.isEmpty() == true) {
34            if ('\n' !in element.text && PsiTreeUtil.findChildOfType(element, PsiComment::class.java) == null) {
35                emit(element.textOffset, "Expand the empty lambda", true).ifAutocorrectAllowed {
36                    element.rightCurlyBrace?.upsertWhitespaceBeforeMe("\n")
37                }
38            }
39        }
40
41        if (element is KtCallExpression && emptyParentheses && element.valueArgumentList == null && element.lambdaArguments.size == 1) {
42            val anchor = element.typeArgumentList ?: element.calleeExpression ?: return
43            emit(element.textOffset, "Use empty parentheses before the trailing lambda", true).ifAutocorrectAllowed {
44                val args = KtPsiFactory(element.project).createCallArguments("()")
45                node.addChild(args.node, anchor.node.treeNext)
46            }
47        }
48        if (element is KtBlockExpression && blankFirst && element.parent is KtNamedFunction) {
49            val first = element.statements.firstOrNull() ?: return
50            // Do not move a leading comment or rewrite its relation to the first statement.
51            if (element.children.any { it is PsiComment && it.textOffset < first.textOffset }) return
52            val leaf = first.node.firstChildLeafOrSelf20
53            val whitespace = leaf.prevLeaf?.takeIf { it.elementType == ElementType.WHITE_SPACE }
54            val old = whitespace?.text.orEmpty()
55            if (old.count { it == '\n' } != 2) {
56                emit(first.textOffset, "Separate the first statement with one blank line", true).ifAutocorrectAllowed {
57                    leaf.upsertWhitespaceBeforeMe("\n\n" + old.substringAfterLast('\n').takeWhile { it == ' ' || it == '\t' })
58                }
59            }
60        }
61        if (element is KtParameter && parameterAnnotation && element.annotationEntries.isNotEmpty()) {
62            if (PsiTreeUtil.findChildOfType(element, PsiComment::class.java) != null) return
63            val annotation = element.annotationEntries.last()
64            val following = annotation.node.lastChildLeafOrSelf20.nextLeaf ?: return
65            if (following.elementType == ElementType.WHITE_SPACE && '\n' in following.text) return
66            emit(annotation.textOffset, "Move the parameter after its annotation", true).ifAutocorrectAllowed {
67                val text = element.containingFile.text
68                val lineStart = text.lastIndexOf('\n', annotation.textOffset - 1) + 1
69                val indent = text.substring(lineStart, annotation.textOffset).takeWhile { it == ' ' || it == '\t' }
70                val desired = "\n" + indent
71                if (following.elementType == ElementType.WHITE_SPACE) following.replaceTextWith(desired)
72                else following.upsertWhitespaceBeforeMe(desired)
73            }
74        }
75        if (element is KtEnumEntry && enumAnnotation && element.annotationEntries.size == 1) {
76            if (PsiTreeUtil.findChildOfType(element, PsiComment::class.java) != null) return
77            val whitespace = element.node.prevLeaf?.takeIf { it.elementType == ElementType.WHITE_SPACE }
78            if (whitespace != null && whitespace.text.count { it == '\n' } > 1) {
79                emit(element.textOffset, "Use one newline between annotated enum entries", true).ifAutocorrectAllowed {
80                    whitespace.replaceTextWith("\n" + whitespace.text.substringAfterLast('\n'))
81                }
82            }
83            val name = element.nameIdentifier ?: return
84            val previous = name.node.firstChildLeafOrSelf20.prevLeaf
85            if (previous?.elementType == ElementType.WHITE_SPACE && previous.text == " ") return
86            emit(name.textOffset, "Keep the enum annotation with its entry", true).ifAutocorrectAllowed {
87                name.node.firstChildLeafOrSelf20.upsertWhitespaceBeforeMe(" ")
88            }
89        }
90    }

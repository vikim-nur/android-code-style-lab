14/** Preserve accepted inheritance spacing without disabling spacing for property/parameter types. */
// Здесь отдельно регулируется пробел перед двоеточием наследования.15class TypeColonSpacingRule : Rule(RuleId("nur:type-colon-spacing"), About(), usesEditorConfigProperties = setOf(INHERITANCE_COLON_SPACE)), RuleAutocorrectApproveHandler {
16    private var inheritanceSpace = true
17    override fun beforeFirstNode(editorConfig: EditorConfig) { inheritanceSpace = editorConfig[INHERITANCE_COLON_SPACE] }
18    private val delegate = SpacingAroundColonRule()
19    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
20        if (node.elementType == ElementType.COLON && node.psi.parent is KtClassOrObject) {
21            val previous = node.prevLeaf
22            val whitespace = previous?.takeIf { it.elementType == ElementType.WHITE_SPACE }
23            if (whitespace?.prevLeaf?.psi is PsiComment) return
24            val desired = if (inheritanceSpace) " " else ""
25            if (whitespace?.text.orEmpty() != desired) {
26                emit(node.startOffset, "Use the selected inheritance colon spacing", true).ifAutocorrectAllowed {
27                    if (desired.isEmpty()) whitespace?.remove() else node.upsertWhitespaceBeforeMe(desired)
28                }
29            }
30            return
31        }
32        delegate.beforeVisitChildNodes(node, emit)
33    }
34}

36/** Keep a single nested call with a trailing lambda compact; retain stock wrapping elsewhere. */
// Обрабатываем поддержанные варианты вложенных лямбд; это не изменение их логики.37class NestedLambdaWrappingRule : Rule(
38    RuleId("nur:nested-lambda-wrapping"), About(),
39    usesEditorConfigProperties = WrappingRule().usesEditorConfigProperties + setOf(FIRST_BASE_INLINE, PARAMETER_ANNOTATION_NEWLINE),
40    visitorModifiers = WrappingRule().visitorModifiers,
41), RuleAutocorrectApproveHandler {
42    private var firstInline = false
43    private var separateParameterAnnotations = false
44    private val delegate = WrappingRule()
45    override fun beforeFirstNode(editorConfig: EditorConfig) {
46        firstInline = editorConfig[FIRST_BASE_INLINE]
47        separateParameterAnnotations = editorConfig[PARAMETER_ANNOTATION_NEWLINE]
48        delegate.beforeFirstNode(editorConfig)
49    }
50    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
51        if (separateParameterAnnotations) {
52            var scope = node.psi as? org.jetbrains.kotlin.com.intellij.psi.PsiElement
53            while (scope != null && scope !is KtClass && scope !is KtNamedFunction) scope = scope.parent
54            val parameters = when (scope) {
55                is KtClass -> scope.primaryConstructorParameters
56                is KtNamedFunction -> scope.valueParameters
57                else -> emptyList()
58            }
59            val headerEnd = when (scope) {
60                is KtClass -> scope.body?.lBrace?.textOffset ?: scope.textRange.endOffset
61                is KtNamedFunction -> scope.bodyExpression?.textOffset ?: scope.textRange.endOffset
62                else -> 0
63            }
64            if (parameters.any { it.annotationEntries.isNotEmpty() } && node.startOffset < headerEnd) return
65        }
66        if (firstInline) {
67            var owner = node.psi as? org.jetbrains.kotlin.com.intellij.psi.PsiElement
68            while (owner != null && owner !is KtClass) owner = owner.parent
69            if (owner is KtClass && owner.superTypeListEntries.size > 1 &&
70                node.startOffset < (owner.body?.lBrace?.textOffset ?: owner.textRange.endOffset)) return
71        }
72        if (firstInline && node.elementType == ElementType.COLON && node.psi.parent is KtClass) return
73        if (node.elementType == ElementType.LPAR) {
74            val list = node.psi.parent as? KtValueArgumentList
75            val argument = list?.arguments?.singleOrNull()
76            val nested = argument?.getArgumentExpression() as? KtCallExpression
77            if (argument != null && !argument.isNamed() && nested != null && nested.lambdaArguments.size == 1) {
78                val lambda = nested.lambdaArguments.single().getLambdaExpression()
79                val head = nested.text.substringBefore('{')
80                // Preserve only the observed layout; do not collapse intentionally expanded outer calls.
81                if (lambda != null && '\n' !in head && list.text.startsWith("(" + nested.calleeExpression?.text) &&
82                    nested.text.endsWith("}") && list.text.endsWith("})")) return
83            }
84        }
85        delegate.beforeVisitChildNodes(node, emit)
86    }
87}

89/** Optional candidate: force a multiline call at 4+ parenthesized arguments. */
// Переносим аргументы подходящего вызова, когда достигнут заданный порог.90class CallArgumentsRule : Rule(
91    RuleId("nur:call-arguments"), About(), usesEditorConfigProperties = setOf(CALL_MIN_ARGUMENTS),
92), RuleAutocorrectApproveHandler {
93    private var minimumArguments = 4
94    override fun beforeFirstNode(editorConfig: EditorConfig) { minimumArguments = editorConfig[CALL_MIN_ARGUMENTS] }
95    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
96        val list = node.psi as? KtValueArgumentList ?: return
97        if (list.parent !is KtCallExpression || list.arguments.size < minimumArguments) return
98        for (argument in list.arguments) {
99            val prev = argument.prevSibling
100            if (prev is PsiWhiteSpace && '\n' in prev.text) continue
101            emit(argument.textOffset, "Place call arguments on separate lines", true)
102                .ifAutocorrectAllowed { argument.node.upsertWhitespaceBeforeMe("\n") }
103        }
104        val right = list.rightParenthesis ?: return
105        if (right.prevSibling !is PsiWhiteSpace || '\n' !in right.prevSibling.text) {
106            emit(right.textOffset, "Place closing parenthesis on a separate line", true)
107                .ifAutocorrectAllowed { right.node.upsertWhitespaceBeforeMe("\n") }
108        }
109    }
110}

112/** Optional candidate: compact only a comment-free one-statement intent lambda within 120 columns. */
// Ищем короткий вызов intent и оформляем его согласно выбранному варианту.113class CompactIntentRule : Rule(
114    RuleId("nur:compact-intent"), About(),
115    usesEditorConfigProperties = setOf(MAX_LINE_LENGTH_PROPERTY, COMPACT_INTENT_MAX_WIDTH, INTENT_EXPANDED),
116), RuleAutocorrectApproveHandler {
117    private var limit = 120
118    private var expanded = false
119    override fun beforeFirstNode(editorConfig: EditorConfig) {
120        limit = minOf(editorConfig[MAX_LINE_LENGTH_PROPERTY], editorConfig[COMPACT_INTENT_MAX_WIDTH])
121        expanded = editorConfig[INTENT_EXPANDED]
122    }
123    override fun afterVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
124        val lambda = node.psi as? KtLambdaExpression ?: return
125        val argument = lambda.parent as? KtLambdaArgument ?: return
126        val call = argument.parent as? KtCallExpression ?: return
// Сравниваем имя вызова с intent. Это ещё не доказательство, что вызов принадлежит Orbit.127        if (call.calleeExpression?.text != "intent") return
128        val block = lambda.bodyExpression ?: return
// Продолжаем только когда внутри ровно одно действие.129        val statement = block.statements.singleOrNull() ?: return
130        if ((!expanded && '\n' !in lambda.text) || '\n' in statement.text) return
131        if (PsiTreeUtil.findChildOfType(lambda, PsiComment::class.java) != null) return
132        val literal = lambda.functionLiteral
133        // Explicit lambda parameters and semantic rewrites are deliberately excluded.
134        if (literal.arrow != null) return
135        if (expanded) {
136            if (literal.arrow != null || '\n' in lambda.text) return
137            emit(lambda.textOffset, "Expand the short intent body", true).ifAutocorrectAllowed {
138                statement.node.firstChildLeafOrSelf20.upsertWhitespaceBeforeMe("\n")
139                lambda.rightCurlyBrace?.upsertWhitespaceBeforeMe("\n")
140            }
141            return
142        }
143        val proposed = "{ " + statement.text + " }"
144        val text = lambda.containingFile.text
145        val lineStart = text.lastIndexOf('\n', lambda.textOffset - 1) + 1
146        val lineEnd = text.indexOf('\n', lambda.textRange.endOffset).let { if (it < 0) text.length else it }
// Учитываем длину всей строки. Если она слишком длинная, ничего не собираем.147        if (lambda.textOffset - lineStart + proposed.length + lineEnd - lambda.textRange.endOffset > limit) return
148        emit(lambda.textOffset, "Keep a short single-statement intent lambda on one line", true).ifAutocorrectAllowed {
149            val before = statement.node.firstChildLeafOrSelf20.prevLeaf
150            val after = statement.node.lastChildLeafOrSelf20.nextLeaf
151            // Only replace boundary whitespace. Never replace or reparse the expression itself.
152            if (before?.elementType == ElementType.WHITE_SPACE) before.replaceTextWith(" ")
153            if (after?.elementType == ElementType.WHITE_SPACE) after.replaceTextWith(" ")
154        }
155    }
156}

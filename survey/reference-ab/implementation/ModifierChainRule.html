8import com.pinterest.ktlint.rule.engine.core.api.editorconfig.EditorConfig

10/** Experimental syntax-only rule. Does not resolve Kotlin types or compile application code. */
11class ModifierChainRule : Rule(
12    ruleId = RuleId("nur:modifier-chain"),
13    about = About(),
14    usesEditorConfigProperties = setOf(MODIFIER_MIN_CALLS, COMPACT_SHORT_MODIFIER, com.pinterest.ktlint.rule.engine.core.api.editorconfig.MAX_LINE_LENGTH_PROPERTY),
15), RuleAutocorrectApproveHandler {
16    private var minimumCalls = 2
17    private var compactShort = false
18    private var maxWidth = 140
19    override fun beforeFirstNode(editorConfig: EditorConfig) {
20        minimumCalls = editorConfig[MODIFIER_MIN_CALLS]
21        compactShort = editorConfig[COMPACT_SHORT_MODIFIER]
22        maxWidth = editorConfig[com.pinterest.ktlint.rule.engine.core.api.editorconfig.MAX_LINE_LENGTH_PROPERTY]
23    }
24    override fun beforeVisitChildNodes(
25        node: ASTNode,
26        emit: (Int, String, Boolean) -> AutocorrectDecision,
27    ) {
28        val expression = node.psi as? KtDotQualifiedExpression ?: return
29        // Only process the outer expression; argument expressions are independent chains.
30        val parent = expression.parent as? KtDotQualifiedExpression
31        if (parent?.receiverExpression == expression) return
32        val chain = mutableListOf<KtDotQualifiedExpression>()
33        var receiver: KtExpression = expression
34        while (receiver is KtDotQualifiedExpression) {
35            // Properties and safe calls are outside this first prototype.
36            if (receiver.selectorExpression !is KtCallExpression) return
37            chain.add(receiver)
38            receiver = receiver.receiverExpression
39        }
40        if (!isKnownModifier(receiver)) return
41        if (chain.size < minimumCalls) {
42            if (!compactShort || org.jetbrains.kotlin.com.intellij.psi.util.PsiTreeUtil.findChildOfType(expression,
43                org.jetbrains.kotlin.com.intellij.psi.PsiComment::class.java) != null) return
44            if (chain.any { '\n' in it.selectorExpression!!.text }) return
45            val dots = chain.mapNotNull { it.node.findChildByType(ElementType.DOT) }
46            val removed = dots.sumOf { dot -> dot.treePrev?.takeIf { it.elementType == ElementType.WHITE_SPACE }?.textLength ?: 0 }
47            val text = expression.containingFile.text
48            val start = text.lastIndexOf('\n', receiver.textOffset - 1) + 1
49            val end = text.indexOf('\n', expression.textRange.endOffset).let { if (it < 0) text.length else it }
50            if (maxWidth > 0 && receiver.textOffset - start + expression.textLength - removed + end - expression.textRange.endOffset > maxWidth) return
51            for (dot in dots) {
52                val whitespace = dot.treePrev?.takeIf { it.elementType == ElementType.WHITE_SPACE } ?: continue
53                emit(dot.startOffset, "Compact the short Modifier chain", true).ifAutocorrectAllowed { whitespace.remove() }
54            }
55            return
56        }
57        for (part in chain.asReversed()) {
58            val dot = part.node.findChildByType(ElementType.DOT) ?: continue
59            val whitespace = dot.treePrev
60            if (whitespace?.elementType == ElementType.WHITE_SPACE && '\n' in whitespace.text) continue
61            emit(dot.startOffset, "Place each Modifier call on a separate line", true)
62                .ifAutocorrectAllowed { dot.upsertWhitespaceBeforeMe("\n") }
63        }
64    }
65
66    private fun isKnownModifier(receiver: KtExpression): Boolean {
67        val name = (receiver as? KtNameReferenceExpression)?.getReferencedName() ?: return false
68        val file = receiver.containingFile as? KtFile ?: return false
69        val importedNames = file.importDirectives.filter {
70            it.importedFqName?.asString() == "androidx.compose.ui.Modifier" && !it.isAllUnder
71        }.map { it.aliasName ?: "Modifier" }.toSet()
72        // Conservative shadowing guard: skip ambiguous blocks, including nested lambdas.
73        var ancestor: PsiElement? = receiver.parent
74        var parameter: KtParameter? = null
75        while (ancestor != null && ancestor !is KtFile) {
76            if (ancestor is KtBlockExpression && ancestor.statements.any {
77                    it is KtNamedDeclaration && it.name == name
78                }) return false
79            if (ancestor is KtClassOrObject && ancestor.declarations.any { it.name == name }) return false
80            if (ancestor is KtLambdaExpression) {
81                if (ancestor.valueParameters.any { it.name == name }) return false
82                if (name == "it" && ancestor.valueParameters.isEmpty()) return false
83            }
84            if (ancestor is KtNamedFunction) {
85                val found = ancestor.valueParameters.firstOrNull { it.name == name }
86                if (found != null) { parameter = found; break }
87            }
88            ancestor = ancestor.parent
89        }
90        if (parameter != null) {
91            val type = parameter.typeReference?.text ?: return false
92            return type == "androidx.compose.ui.Modifier" || type in importedNames
93        }
94        // Avoid same-file declarations shadowing the imported singleton.
95        if (file.declarations.any { it.name == name }) return false
96        return name in importedNames
97    }

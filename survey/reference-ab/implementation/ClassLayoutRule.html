12/** Whitespace-only canonical class headers. Ambiguous headers retain their layout. */
13class ClassLayoutRule : Rule(
14    RuleId("nur:class-layout"), About(),
15    usesEditorConfigProperties = setOf(MAX_LINE_LENGTH_PROPERTY, INDENT_SIZE_PROPERTY, CONSTRUCTOR_MIN_PARAMETERS, DATA_CONSTRUCTOR_MIN_PARAMETERS, INJECT_MULTILINE, WRAPPED_BASE_ARGUMENTS, BASE_ARGUMENT_MINIMUM, PLAIN_CONSTRUCTOR_BASE_NEWLINE, FIRST_BASE_INLINE, INHERITANCE_COLON_SPACE, PARAMETER_ANNOTATION_NEWLINE),
16    visitorModifiers = setOf(VisitorModifier.RunAsLateAsPossible,
17        VisitorModifier.RunAfterRule(RuleId("standard:argument-list-wrapping"),
18            VisitorModifier.RunAfterRule.Mode.ONLY_WHEN_RUN_AFTER_RULE_IS_LOADED_AND_ENABLED)),
19), RuleAutocorrectApproveHandler {
20    private var maxWidth = 140
21    private var indentSize = 4
22    private var parameterLimit = 4
23    private var dataParameterLimit = 4
24    private var injectMultiline = true
25    private var wrappedBaseArguments = true
26    private var baseArgumentMinimum = Int.MAX_VALUE
27    private var plainBaseNewline = false
28    private var firstBaseInline = false
29    private var colonSpace = true
30    private var annotationNewline = false
31
32    override fun beforeFirstNode(editorConfig: EditorConfig) {
33        maxWidth = editorConfig[MAX_LINE_LENGTH_PROPERTY].let { if (it <= 0) Int.MAX_VALUE else it }
34        indentSize = editorConfig[INDENT_SIZE_PROPERTY]
35        parameterLimit = editorConfig[CONSTRUCTOR_MIN_PARAMETERS]
36        dataParameterLimit = editorConfig[DATA_CONSTRUCTOR_MIN_PARAMETERS]
37        injectMultiline = editorConfig[INJECT_MULTILINE]
38        wrappedBaseArguments = editorConfig[WRAPPED_BASE_ARGUMENTS]
39        baseArgumentMinimum = editorConfig[BASE_ARGUMENT_MINIMUM]
40        plainBaseNewline = editorConfig[PLAIN_CONSTRUCTOR_BASE_NEWLINE]
41        firstBaseInline = editorConfig[FIRST_BASE_INLINE]
42        colonSpace = editorConfig[INHERITANCE_COLON_SPACE]
43        annotationNewline = editorConfig[PARAMETER_ANNOTATION_NEWLINE]
44    }
45
46    override fun beforeVisitChildNodes(node: ASTNode, emit: (Int, String, Boolean) -> AutocorrectDecision) {
47        val declaration = node.psi as? KtClass ?: return
48        if (declaration is KtEnumEntry) return
49        val keyword = declaration.getDeclarationKeyword() ?: return
50        val bodyBrace = declaration.body?.lBrace
51        val start = keyword.textOffset
52        val end = bodyBrace?.textOffset ?: declaration.textRange.endOffset
53        val headerElements = descendants(declaration).filter { it.textOffset >= start && it.textRange.endOffset <= end }
54        // Retain comments, multiline literals/default expressions, and where clauses. No text reparsing.
55        if (declaration.typeConstraintList != null || headerElements.any {
56                it is PsiComment || it is KtLambdaExpression || it is KtObjectLiteralExpression ||
57                    (it is KtStringTemplateExpression && '\n' in it.text)
58            }) {
59            return
60        }
61        val leaves = mutableListOf<ASTNode>()
62        fun collect(current: ASTNode) {
63            if (current.startOffset >= end || current.startOffset + current.textLength <= start) return
64            if (current.firstChildNode == null) leaves.add(current)
65            else { var child = current.firstChildNode; while (child != null) { collect(child); child = child.treeNext } }
66        }
67        collect(node)
68        val code = leaves.filter { it.elementType != ElementType.WHITE_SPACE }
69        if (code.isEmpty()) return
70        val fullText = declaration.containingFile.text
71        // An empty body contributes " {}", not just " {". Include any body text
72        // that remains on the header line, so the width decision is repeatable.
73        val bodySuffixWidth = bodyBrace?.let {
74            fullText.substring(it.textOffset, declaration.textRange.endOffset)
75                .substringBefore('\n').trimEnd().length + 1
76        } ?: 0
77        val lineStart = fullText.lastIndexOf('\n', start - 1) + 1
78        val prefix = fullText.substring(lineStart, start)
79        val baseIndent = prefix.takeWhile { it == ' ' || it == '\t' }
80        val oneIndent = " ".repeat(indentSize)
81        val desired = linkedMapOf<ASTNode, String>()
82        for (leaf in code.drop(1)) {
83            val whitespace = leaf.prevLeaf?.takeIf { it.elementType == ElementType.WHITE_SPACE }
84            if (whitespace != null) desired[leaf] = if ('\n' in whitespace.text) " " else whitespace.text
85        }
86        fun gap(element: PsiElement?, text: String) { if (element != null) desired[element.node.firstChildLeafOrSelf20] = text }
87        fun inlineLength(element: PsiElement): Int {
88            val tokens = code.filter {
89                it.startOffset >= element.textOffset && it.startOffset < element.textRange.endOffset
90            }
91            return tokens.mapIndexed { index, leaf ->
92                (if (index == 0) "" else desired[leaf].orEmpty()) + leaf.text
93            }.joinToString("").length
94        }
95        val parameters = declaration.getPrimaryConstructorParameterList()
96        val values = parameters?.parameters.orEmpty()
97        val supers = declaration.superTypeListEntries
98        val colon = declaration.getColon()
99        val inject = declaration.primaryConstructor?.annotationEntries.orEmpty().any {
100            it.shortName?.asString() == "Inject"
101        }
102        val signatureEnd = colon?.textOffset ?: end
103        val primaryText = fullText.substring(start, signatureEnd).replace(Regex("\\s+"), " ").trim()
104        val multilineParameters = values.isNotEmpty() && (
105            injectMultiline && inject || values.size >= (if (declaration.isData()) dataParameterLimit else parameterLimit) ||
106                prefix.length + primaryText.length + 2 > maxWidth ||
107                values.any { '\n' in it.text })
108        if (parameters != null) {
109            for (parameter in values) {
110                gap(parameter, if (multilineParameters) "\n$baseIndent$oneIndent" else if (parameter == values.first()) "" else " ")
111            }
112            gap(parameters.rightParenthesis, if (multilineParameters) "\n$baseIndent" else "")
113        }
114        // Do not erase annotations inside individual parameters; they retain their own wrapping.
115        for (parameter in values) {
116            for (leaf in code) {
117                if (leaf.startOffset > (if (annotationNewline) parameter.textRange.startOffset else parameter.textOffset) && leaf.startOffset < parameter.textRange.endOffset) {
118                    val old = leaf.prevLeaf?.takeIf { it.elementType == ElementType.WHITE_SPACE }
119                    if (old != null && '\n' in old.text) desired[leaf] = old.text
120                }
121            }
122        }
123        fun computedHeaderLength(stopAt: Int = end, suffix: Int = bodySuffixWidth): Int {
124            val firstLine = StringBuilder(prefix)
125            for ((i, leaf) in code.withIndex()) {
126                if (leaf.startOffset >= stopAt) break
127                if (i > 0) firstLine.append(desired[leaf].orEmpty())
128                firstLine.append(leaf.text)
129            }
130            return firstLine.toString().substringAfterLast('\n').length + suffix
131        }
132        // Measure the prospective inline layout, including gaps that earlier rules may
133        // have removed while the superclass was on another line.
134        gap(colon, if (colonSpace) " " else "")
135        for (entry in supers) {
136            gap(entry, " ")
137            val args = (entry as? KtSuperTypeCallEntry)?.valueArgumentList ?: continue
138            for ((index, argument) in args.arguments.withIndex()) gap(argument, if (index == 0) "" else " ")
139            gap(args.rightParenthesis, "")
140        }
141        val inlineFirst = firstBaseInline && supers.size > 1 &&
142            computedHeaderLength(supers.first().textRange.endOffset, 1) <= maxWidth
143        val forceSingleNewline = plainBaseNewline && !inject && values.isNotEmpty() && supers.size == 1
144        val wrapSupers = forceSingleNewline || supers.size > 1 || computedHeaderLength() > maxWidth
145        for ((index, entry) in supers.withIndex()) {
146            val entryWrapped = index > 0 || wrapSupers && !inlineFirst
147            gap(entry, if (!entryWrapped) " " else "\n$baseIndent$oneIndent")
148            val args = (entry as? KtSuperTypeCallEntry)?.valueArgumentList ?: continue
149            val entryIndent = baseIndent + if (entryWrapped) oneIndent else ""
150            val entryStartWidth = if (entryWrapped) entryIndent.length else {
151                val text = StringBuilder(prefix)
152                for ((i, leaf) in code.withIndex()) {
153                    if (leaf.startOffset >= entry.textOffset) break
154                    if (i > 0) text.append(desired[leaf].orEmpty())
155                    text.append(leaf.text)
156                }
157                text.toString().substringAfterLast('\n').length + 1
158            }
159            val wrapArguments = wrappedBaseArguments && entryWrapped && args.arguments.isNotEmpty() || args.arguments.size >= baseArgumentMinimum || entryStartWidth + inlineLength(entry) + bodySuffixWidth > maxWidth
160            for ((argIndex, argument) in args.arguments.withIndex()) {
161                gap(argument, if (wrapArguments) "\n$entryIndent$oneIndent" else if (argIndex == 0) "" else " ")
162            }
163            gap(args.rightParenthesis, if (wrapArguments && args.arguments.isNotEmpty()) "\n$entryIndent" else "")
164        }
165        gap(bodyBrace, " ")
166        // Reverse order keeps source offsets valid. Only whitespace leaves are changed or inserted.
167        for ((leaf, whitespace) in desired.entries.sortedByDescending { it.key.startOffset }) {
168            val previous = leaf.prevLeaf
169            val actual = if (previous?.elementType == ElementType.WHITE_SPACE) previous.text else ""
170            if (actual != whitespace) {
171                emit(leaf.startOffset, "Use the shared class header layout", true).ifAutocorrectAllowed {
172                    if (whitespace.isEmpty() && previous?.elementType == ElementType.WHITE_SPACE) previous.remove()
173                    else leaf.upsertWhitespaceBeforeMe(whitespace)
174                }
175            }
176        }
177    }
178
179    private fun descendants(root: PsiElement): List<PsiElement> =
180        PsiTreeUtil.collectElements(root) { true }.toList()
